### Скрипты нагрузки (k6)

#### 1. `load/diff_review.js` (Потоковая отправка валидных diff и сценарий деградации)
```javascript
import http from 'k6/http';
import { check, sleep } from 'k6';

export const options = {
  vus: 10,
  duration: '120s',
  thresholds: {
    // REL-1: таймаут 10с + запас на валидацию и маскирование; 0% серверных ошибок
    http_req_duration: ['p(95)<10500'],
    http_req_failed: ['rate==0'],
  },
};

const payload = JSON.stringify({
  diff: 'diff --git a/app.py b/app.py\n' + '+// valid diff content\n'.repeat(250), // ~5000 символов
});

const params = {
  headers: {
    'Content-Type': 'application/json',
    'X-Mock-LLM-Latency-Ms': __ENV.MOCK_LATENCY || '200', // Для теста деградации: -e MOCK_LATENCY=11000
  },
};

export default function () {
  const res = http.post('http://localhost:8000/api/reviews', payload, params);

  if (__ENV.MOCK_LATENCY === '11000') {
    // Ожидаем срабатывание REL-1 (контролируемый таймаут 10с)
    check(res, {
      'status is 504 on LLM timeout': (r) => r.status === 504,
    });
  } else {
    check(res, {
      'status is 200': (r) => r.status === 200,
      'has summary': (r) => JSON.parse(r.body).summary !== undefined,
    });
  }
  sleep(1);
}
```

Команды запуска:
- Штатный прогон: `k6 run load/diff_review.js`
- Проверка сценария деградации (задержка LLM 11с > таймаута `REL-1` 10с): `k6 run -e MOCK_LATENCY=11000 load/diff_review.js`

---

На данном этапе MVP сервис не рассчитан на публичную неконтролируемую нагрузку, так как узким местом является внешний API LLM с жесткими лимитами RPS и стоимостью токенов. Полномасштабное стресс-тестирование с тысячами запросов потребуется при переходе сервиса в общекорпоративный CI/CD конвейер и подключении пула локальных LLM-воркеров.
